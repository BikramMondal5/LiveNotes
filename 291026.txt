-- 3(i) Extract the student who has registered to DBMS with the name "Anil Sharma".

FROM Student s JOIN Enrolled e ON s.sid = e.sid JOIN Course c ON c.cid = e.cid

-- 3(ii) Get all courses being taught in a semester.
SELECT cid, cname, fee FROM Course WHERE Semester = &semester;

-- 3(iii) List students along with the courses they are enrolled in.
SELECT s.sid, s.sname, c.cid, c.cname
FROM Student s JOIN Enrolled e ON s.sid = e.sid JOIN Course c ON c.cid = e.cid
ORDER BY s.sid;

-- 3(iv) List all students who enrolled after a certain date (e.g., 01-JUL-2025).
SELECT s.sid, s.sname, c.cname, e.doj
FROM Student s JOIN Enrolled e ON s.sid = e.sid JOIN Course c ON c.cid = e.cid
WHERE e.doj > TO_DATE('01-JUL-2025','DD-MON-YYYY');

-- 3(v) Find the number of students enrolled in each course.
SELECT c.cid, c.cname, COUNT(e.sid) AS Students_Enrolled
FROM Course c LEFT JOIN Enrolled e ON c.cid = e.cid GROUP BY c.cid, c.cname;

-- 3(vi) Get the total fee collected per course.
SELECT c.cid, c.cname, NVL(SUM(e.Fee_Paid),0) AS Total_Fee_Collected
FROM Course c LEFT JOIN Enrolled e ON c.cid = e.cid GROUP BY c.cid, c.cname;

-- 3(vii) Get the list of enrolled students who need to pay the fee.
SELECT s.sid, s.sname, c.cname, c.fee, e.Fee_Paid, (c.fee - e.Fee_Paid) AS Balance_Due
FROM Student s JOIN Enrolled e ON s.sid = e.sid JOIN Course c ON c.cid = e.cid
WHERE e.Fee_Paid < c.fee;

-- STEP 1 (1NF): remove repeating groups - every cell atomic; one row per
--               (supplier, part, project). Key = (Supplier_ID, Part_ID, Project_ID)
CREATE TABLE Supply_Detail_1NF AS
SELECT u.Supplier_ID, u.Supplier_Name, u.Supplier_City,
       TRIM(REGEXP_SUBSTR(u.Part_List,    '[^,]+', 1, lv.n)) AS Part_ID,
       TRIM(REGEXP_SUBSTR(u.Project_List, '[^,]+', 1, lv.n)) AS Project_ID,
       TO_NUMBER(TRIM(REGEXP_SUBSTR(u.Qty_List, '[^,]+', 1, lv.n))) AS Qty
FROM Supply_Detail_UNF u
JOIN (SELECT LEVEL AS n FROM dual CONNECT BY LEVEL <= 100) lv
  ON lv.n <= REGEXP_COUNT(u.Part_List, ',') + 1;
ALTER TABLE Supply_Detail_1NF ADD CONSTRAINT pk_sd1nf PRIMARY KEY (Supplier_ID, Part_ID, Project_ID);

CREATE OR REPLACE PROCEDURE emp_category (p_empid IN Employee.EmpID%TYPE) IS
  v_fname Employee.EmpFname%TYPE;
  v_lname Employee.EmpLname%TYPE;
  v_age   Employee.Age%TYPE;
  v_cat   VARCHAR2(20);

END emp_category;

-- 9.2 Function: FOR loop (factorial)
CREATE OR REPLACE FUNCTION fact (n IN NUMBER) RETURN NUMBER IS
  f NUMBER := 1;
BEGIN
  FOR i IN 1 .. n LOOP f := f * i; END LOOP;
  RETURN f;
END fact;

-- 9.3 Function: WHILE loop (prime check)
CREATE OR REPLACE FUNCTION is_prime (n IN NUMBER) RETURN VARCHAR2 IS
  i NUMBER := 2;
BEGIN
  IF n < 2 THEN RETURN 'NO'; END IF;
  WHILE i * i <= n LOOP
    IF MOD(n, i) = 0 THEN RETURN 'NO'; END IF;
    i := i + 1;
  END LOOP;
  RETURN 'YES';
END is_prime;
/

-- 9.4 Procedure: basic LOOP with EXIT WHEN (Fibonacci)
CREATE OR REPLACE PROCEDURE fibonacci (p_terms IN NUMBER) IS
  a NUMBER := 0; b NUMBER := 1; c NUMBER; k NUMBER := 1; s VARCHAR2(500) := '';
BEGIN
  LOOP
    EXIT WHEN k > p_terms;
    s := s || a || ' ';
    c := a + b; a := b; b := c; k := k + 1;
  END LOOP;
  DBMS_OUTPUT.PUT_LINE(s);
END fibonacci;
/

                         WHEN 'C' THEN 'Average' ELSE 'Needs improvement' END;
  DBMS_OUTPUT.PUT_LINE('Grade ' || v_grade || ' = ' || v_text);
  FOR i IN 1 .. 10 LOOP DBMS_OUTPUT.PUT_LINE('7 x ' || i || ' = ' || 7 * i); END LOOP;

-- ######################################################################
-- PROBLEM STATEMENT 10: Cursors and Triggers
-- Q: Creating cursors, parameterized cursor, locks on cursors, advantages of
--    cursors. Introduction to triggers - constraints vs triggers, creating,
--    altering.
-- (Uses Employee table of Problem 2 and Book table of Problem 5.)

SET SERVEROUTPUT ON

-- 10.1 Implicit cursor (SQL%FOUND, SQL%ROWCOUNT)
BEGIN
  UPDATE Employee SET Age = Age + 1 WHERE City = 'Kolkata';
  IF SQL%FOUND THEN DBMS_OUTPUT.PUT_LINE(SQL%ROWCOUNT || ' row(s) updated');
  ELSE DBMS_OUTPUT.PUT_LINE('No rows matched');
  END IF;
  ROLLBACK;
END;
/

-- 10.2 Explicit cursor: OPEN, FETCH, CLOSE
DECLARE
  CURSOR c_emp IS SELECT EmpID, EmpFname, EmpLname, City FROM Employee ORDER BY EmpID;
  v_id Employee.EmpID%TYPE; v_fn Employee.EmpFname%TYPE;
  v_ln Employee.EmpLname%TYPE; v_city Employee.City%TYPE;
BEGIN
  OPEN c_emp;
  LOOP
    FETCH c_emp INTO v_id, v_fn, v_ln, v_city;
    EXIT WHEN c_emp%NOTFOUND;
    DBMS_OUTPUT.PUT_LINE(v_id || ' - ' || v_fn || ' ' || v_ln || ' (' || v_city || ')');
  END LOOP;
  DBMS_OUTPUT.PUT_LINE('Rows fetched: ' || c_emp%ROWCOUNT);
  CLOSE c_emp;
END;
/

-- 10.3 Cursor FOR loop (auto open/fetch/close)
BEGIN
  FOR r IN (SELECT EmpFname, EmpLname, Age FROM Employee WHERE Age > 40) LOOP
    DBMS_OUTPUT.PUT_LINE(r.EmpFname || ' ' || r.EmpLname || ' - ' || r.Age);
  END LOOP;
END;
/

-- 10.4 Parameterized cursor
DECLARE
  CURSOR c_city (p_city Employee.City%TYPE, p_age Employee.Age%TYPE DEFAULT 0) IS
    SELECT EmpFname, EmpLname, Age FROM Employee WHERE City = p_city AND Age >= p_age;
BEGIN
  FOR r IN c_city('Kolkata') LOOP
    DBMS_OUTPUT.PUT_LINE('Kolkata: ' || r.EmpFname || ' ' || r.EmpLname);
  END LOOP;
  FOR r IN c_city('Mumbai', 40) LOOP
    DBMS_OUTPUT.PUT_LINE('Mumbai 40+: ' || r.EmpFname || ' ' || r.EmpLname);
  END LOOP;
END;
/

-- 10.5 Locks on cursors: FOR UPDATE ... NOWAIT and WHERE CURRENT OF
--      (rows stay locked until COMMIT / ROLLBACK)
DECLARE
  CURSOR c_book IS
    SELECT BookID, Title, Price FROM Book WHERE Genre = 'Fiction' FOR UPDATE OF Price NOWAIT;
BEGIN
  FOR r IN c_book LOOP
    UPDATE Book SET Price = ROUND(r.Price * 1.05, 2) WHERE CURRENT OF c_book;
    DBMS_OUTPUT.PUT_LINE(r.Title || ': ' || r.Price || ' -> ' || ROUND(r.Price * 1.05, 2));
  END LOOP;
  COMMIT;
EXCEPTION
  WHEN OTHERS THEN ROLLBACK; DBMS_OUTPUT.PUT_LINE('Could not lock rows: ' || SQLERRM);
END;
/

-- 10.6 REF CURSOR (cursor variable) returned by a function
CREATE OR REPLACE FUNCTION emp_by_city (p_city IN VARCHAR2) RETURN SYS_REFCURSOR IS
  rc SYS_REFCURSOR;
BEGIN
  OPEN rc FOR SELECT EmpID, EmpFname, EmpLname FROM Employee WHERE City = p_city;
  RETURN rc;
END;
/
DECLARE
  rc SYS_REFCURSOR; v_id Employee.EmpID%TYPE; v_fn Employee.EmpFname%TYPE; v_ln Employee.EmpLname%TYPE;
BEGIN
  rc := emp_by_city('Delhi');
  LOOP
    FETCH rc INTO v_id, v_fn, v_ln;
    EXIT WHEN rc%NOTFOUND;
    DBMS_OUTPUT.PUT_LINE(v_id || ' ' || v_fn || ' ' || v_ln);
  END LOOP;
  CLOSE rc;
END;
/

-- 10.7 Advantages of cursors:
--  * row-by-row processing of a query result with IF / loops
--  * moves SQL results into PL/SQL variables for calculations
--  * parameterized cursors give reuse with different values
--  * FOR UPDATE / WHERE CURRENT OF give row locking and safe updates
--  * %FOUND, %NOTFOUND, %ROWCOUNT, %ISOPEN control the processing
--  * rows are fetched on demand (BULK COLLECT ... LIMIT for speed)
--  Drawback: slower than one set-based SQL statement; use only when needed.

-- 10.8 Constraints vs Triggers:
--  Constraint: declarative rule (PK, FK, UNIQUE, NOT NULL, CHECK), fast, checks all rows.
--  Trigger   : PL/SQL block fired by an event (INSERT/UPDATE/DELETE), any logic,
--              audit, derived values, cross-table rules; slower, fires only on the event.
--  Use constraints when the rule can be declared; triggers when it cannot.

-- 10.9 Creating triggers
-- (a) BEFORE row-level trigger: validate data
CREATE OR REPLACE TRIGGER trg_emp_validate
BEFORE INSERT OR UPDATE OF Age ON Employee
FOR EACH ROW
BEGIN
  IF :NEW.Age < 18 OR :NEW.Age > 65 THEN
    RAISE_APPLICATION_ERROR(-20010, 'Employee age must be between 18 and 65');
  END IF;
  IF INSERTING AND :NEW.EmailID IS NULL THEN
    :NEW.EmailID := LOWER(:NEW.EmpFname || '.' || :NEW.EmpLname) || '@corp.com';
  END IF;
END;
/

-- (b) AFTER row-level audit trigger (sequence used since 11g has no IDENTITY)
CREATE TABLE Emp_Audit (
  Audit_ID NUMBER PRIMARY KEY, Action VARCHAR2(10), EmpID NUMBER(5),
  Old_Age NUMBER(3), New_Age NUMBER(3), Changed_By VARCHAR2(30), Changed_On DATE);
CREATE SEQUENCE seq_emp_audit START WITH 1 INCREMENT BY 1;

CREATE OR REPLACE TRIGGER trg_emp_audit
AFTER INSERT OR UPDATE OR DELETE ON Employee
FOR EACH ROW
DECLARE v_action VARCHAR2(10);
BEGIN
  IF INSERTING THEN v_action := 'INSERT';
  ELSIF UPDATING THEN v_action := 'UPDATE';
  ELSE v_action := 'DELETE';
  END IF;
  INSERT INTO Emp_Audit
  VALUES (seq_emp_audit.NEXTVAL, v_action, NVL(:NEW.EmpID, :OLD.EmpID),
          :OLD.Age, :NEW.Age, USER, SYSDATE);
END;
/
UPDATE Employee SET Age = 53 WHERE EmpID = 102;   -- logged in Emp_Audit
UPDATE Employee SET Age = 10 WHERE EmpID = 102;   -- ORA-20010 from trg_emp_validate
SELECT * FROM Emp_Audit;
ROLLBACK;

-- 10.10 Altering triggers (body changes use CREATE OR REPLACE; ALTER enables,
--       disables, renames or recompiles)
ALTER TRIGGER trg_emp_audit DISABLE;
ALTER TRIGGER trg_emp_audit ENABLE;
ALTER TABLE Employee DISABLE ALL TRIGGERS;
ALTER TABLE Employee ENABLE ALL TRIGGERS;
ALTER TRIGGER trg_emp_validate RENAME TO trg_emp_check;
ALTER TRIGGER trg_emp_check COMPILE;
